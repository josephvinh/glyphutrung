require __DIR__ . '/_bootstrap.php';

$action = $_GET['action'] ?? '';

// Duyệt tài khoản + quản lý nhân sự thuộc module Nhân sự (staff).
// Còn lại (khối, lớp, chủ nhiệm/trưởng khối) thuộc module Tổ chức (org).
$STAFF_ACTIONS = ['saveMember', 'deleteMember', 'approveMember', 'rejectMember', 'resetPassword'];
$me   = in_array($action, $STAFF_ACTIONS, true)
    ? require_permission('staff', 'edit')
    : require_permission('org', 'edit');

$year = current_year();
if (!$year) json_fail('Chưa có niên khoá nào đang mở.', 409);

$yid = (int) $year['id'];

!!! CẢNH BÁO QUAN TRỌNG !!!
Chuỗi TÌM ở chỗ 2, 3, 4 GIỐNG HỆT NHAU.
KHÔNG được dùng Replace All / Thay thế tất cả.
Phải sửa từng chỗ, và nội dung THAY ở mỗi chỗ KHÁC NHAU:

rồi sửa dòng ngay dưới nó. Làm lần lượt 3 case.

================================================================
  KIỂM TRA SAU KHI SỬA
================================================================

1. Đăng nhập BĐH  -> Nhân sự -> thấy hàng chờ duyệt -> bấm Duyệt
   Kỳ vọng: duyệt THÀNH CÔNG.
   (Trước khi sửa: báo "Tài khoản này đã được duyệt rồi.")

2. Đăng nhập Trưởng khối -> Nhân sự
   Kỳ vọng: KHÔNG thấy nút "Duyệt" / "Từ chối".

3. Nếu PHP báo lỗi cú pháp: chạy  php -l public/api/org.php


================================================================
  LƯU Ý VỀ QUYỀN (thay đổi hành vi)
================================================================

Sau khi sửa, Trưởng khối MẤT quyền thêm/sửa nhân sự
(trước đây họ có nhờ quyền 'org'=edit).

Nếu muốn giữ quyền đó cho Trưởng khối, chạy SQL:

    UPDATE permissions SET level='edit'
     WHERE module_key='staff' AND role_code='truong_khoi';
